cmake_minimum_required(VERSION 3.25)
project(AottgInSkyrim VERSION 0.1.0 LANGUAGES C CXX)

set(COMMONLIB_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../tools/SkyrimDev/vendor/CommonLibSSE-NG" CACHE PATH "Pinned CommonLibSSE-NG source")
set(LIBAOTTG_BUILD_ID "local" CACHE STRING "Build identity included in the plugin log")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_subdirectory("${COMMONLIB_PATH}" commonlib)
include("${COMMONLIB_PATH}/cmake/CommonLibSSE.cmake")
target_compile_options(CommonLibSSE PRIVATE /MP4)
set(LIBAOTTG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../LibAottg/native" libaottg-native)

# Declare metadata in Plugin.cpp so SKSE also knows we require post-1.6.629
# structures. The helper's USE_ADDRESS_LIBRARY option cannot express that floor.
add_library(AottgInSkyrim SHARED src/Plugin.cpp src/Character.cpp src/SkyrimWorld.cpp src/Render.cpp src/Runtime.cpp src/Settings.cpp src/SettingsMenu.cpp src/CallDiscovery.cpp)
target_link_libraries(AottgInSkyrim PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_definitions(AottgInSkyrim PRIVATE LIBAOTTG_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} LIBAOTTG_VERSION_MINOR=${PROJECT_VERSION_MINOR} LIBAOTTG_VERSION_PATCH=${PROJECT_VERSION_PATCH})

add_library(LibAottgCallDecoder STATIC third_party/hde64/hde64.c)
target_include_directories(LibAottgCallDecoder PUBLIC third_party/hde64)
target_link_libraries(AottgInSkyrim PRIVATE LibAottgCallDecoder)
add_executable(LibAottgHookCheck tests/HookCheck.cpp)
target_compile_features(LibAottgHookCheck PRIVATE cxx_std_23)
target_include_directories(LibAottgHookCheck PRIVATE src)
target_link_libraries(LibAottgHookCheck PRIVATE LibAottgCallDecoder)

target_include_directories(AottgInSkyrim SYSTEM PRIVATE third_party/skse-menu-framework)
target_precompile_headers(AottgInSkyrim PRIVATE src/PCH.h)
find_package(directxtk CONFIG REQUIRED)
target_link_libraries(AottgInSkyrim PRIVATE LibAottgNative Microsoft::DirectXTK d3d11.lib d3dcompiler.lib dxgi.lib ole32.lib)
target_compile_definitions(AottgInSkyrim PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)

add_executable(LibAottgAssetCheck tests/AssetCheck.cpp src/Character.cpp)
add_executable(LibAottgSettingsCheck tests/SettingsCheck.cpp src/Settings.cpp)
add_executable(LibAottgRenderDepthCheck tests/RenderDepthCheck.cpp)
target_compile_features(LibAottgRenderDepthCheck PRIVATE cxx_std_23)
target_include_directories(LibAottgRenderDepthCheck PRIVATE src)
target_link_libraries(LibAottgRenderDepthCheck PRIVATE d3d11.lib d3dcompiler.lib)
target_compile_features(LibAottgSettingsCheck PRIVATE cxx_std_23)
target_include_directories(LibAottgSettingsCheck PRIVATE src)
target_link_libraries(LibAottgSettingsCheck PRIVATE LibAottgNative)
target_compile_definitions(LibAottgSettingsCheck PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_features(LibAottgAssetCheck PRIVATE cxx_std_23)
target_include_directories(LibAottgAssetCheck PRIVATE src)
target_link_libraries(LibAottgAssetCheck PRIVATE LibAottgNative Microsoft::DirectXTK)
target_compile_definitions(AottgInSkyrim PRIVATE "LIBAOTTG_BUILD_ID=\"${LIBAOTTG_BUILD_ID}\"")
target_compile_options(AottgInSkyrim PRIVATE /W4 /Zi /permissive- /MP4)
target_link_options(AottgInSkyrim PRIVATE /DEBUG:FULL)
install(TARGETS AottgInSkyrim RUNTIME DESTINATION SKSE/Plugins)
install(FILES "$<TARGET_PDB_FILE:AottgInSkyrim>" DESTINATION SKSE/Plugins OPTIONAL)
install(DIRECTORY assets/hero DESTINATION SKSE/Plugins/LibAottg)

install(DIRECTORY assets/ui DESTINATION SKSE/Plugins/LibAottg)
install(DIRECTORY assets/hero/ DESTINATION Sound/LibAottg FILES_MATCHING PATTERN "*.wav")
